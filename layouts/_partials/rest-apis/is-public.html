{{- /* True only when the operation's x-internal audience is "public" or a list that contains the string "public". A missing key, "cloud" alone and every other shape are false, so an operation is public by opt-in only. The exporter (.github/scripts/openapi_export.py) applies the same rule before the file reaches Hugo; this is the second lock. Context: the operation map. */ -}} {{- $audience := index . "x-internal" -}} {{- $public := false -}} {{- if reflect.IsSlice $audience -}} {{- range $audience -}} {{- if and (eq (printf "%T" .) "string") (eq . "public") }}{{ $public = true }}{{ end -}} {{- end -}} {{- else if and (eq (printf "%T" $audience) "string") (eq $audience "public") -}} {{- $public = true -}} {{- end -}} {{- return $public -}}
